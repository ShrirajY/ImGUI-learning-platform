ImGui Cheat Sheet

The calls you'll use most, grouped by job. Version: Dear ImGui 1.92.0 WIP (master branch).

Frame skeleton

// init (after GL context)
IMGUI_CHECKVERSION();
ImGui::CreateContext();
ImGui::StyleColorsDark();
ImGui_ImplWin32_InitForOpenGL(hwnd);
ImGui_ImplOpenGL2_Init();

// every frame
ImGui_ImplOpenGL2_NewFrame();
ImGui_ImplWin32_NewFrame();
ImGui::NewFrame();
    /* ... your UI ... */
ImGui::Render();
ImGui_ImplOpenGL2_RenderDrawData(ImGui::GetDrawData());
SwapBuffers(hdc);

// shutdown (before deleting GL context)
ImGui_ImplOpenGL2_Shutdown();
ImGui_ImplWin32_Shutdown();
ImGui::DestroyContext();
